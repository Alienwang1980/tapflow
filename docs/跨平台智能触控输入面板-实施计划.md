from fastapi import WebSocket, WebSocketDisconnect
import asyncio
import logging

logger = logging.getLogger("stp.connection")

class ConnectionManager:
    """管理所有 WebSocket 客户端连接"""

    def __init__(self):
        self.active: dict[str, WebSocket] = {}  # client_id -> websocket

    async def connect(self, client_id: str, websocket: WebSocket):
        await websocket.accept()
        self.active[client_id] = websocket
        logger.info(f"Client connected: {client_id} (total: {len(self.active)})")

    def disconnect(self, client_id: str):
        if client_id in self.active:
            del self.active[client_id]
        logger.info(f"Client disconnected: {client_id} (total: {len(self.active)})")

    async def send_to(self, client_id: str, message: dict):
        if client_id in self.active:
            try:
                await self.active[client_id].send_json(message)
            except Exception as e:
                logger.warning(f"Send to {client_id} failed: {e}")
                self.disconnect(client_id)

    async def broadcast(self, message: dict):
        dead = []
        for cid, ws in self.active.items():
            try:
                await ws.send_json(message)
            except Exception:
                dead.append(cid)
        for cid in dead:
            self.disconnect(cid)

    @property
    def count(self) -> int:
        return len(self.active)


manager = ConnectionManager()
```

"""
macOS 输入模拟引擎
使用 Quartz.CoreGraphics (CGEvent) API 注入键盘事件
注意: 需要 Accessibility 权限 (System Settings → Privacy → Accessibility)
"""

import logging

logger = logging.getLogger("stp.input")

try:
    from Quartz.CoreGraphics import (
        CGEventCreateKeyboardEvent,
        CGEventPost,
        kCGHIDEventTap,
        CGEventSetFlags,
    )
    from Quartz.CoreGraphics import kCGEventKeyDown, kCGEventKeyUp
    HAVE_QUARTZ = True
except ImportError:
    HAVE_QUARTZ = False
    logger.warning("Quartz import failed — running on non-macOS or missing pyobjc")
    # 开发阶段模拟实现
    kCGHIDEventTap = 0
    kCGEventKeyDown = 10
    kCGEventKeyUp = 11


# macOS 虚拟键码映射表（常用键）
KEYCODE_MAP = {
    # 字母
    'A': 0x00, 'B': 0x0B, 'C': 0x08, 'D': 0x02, 'E': 0x0E,
    'F': 0x03, 'G': 0x05, 'H': 0x04, 'I': 0x22, 'J': 0x26,
    'K': 0x28, 'L': 0x25, 'M': 0x2E, 'N': 0x2D, 'O': 0x1F,
    'P': 0x23, 'Q': 0x0C, 'R': 0x0F, 'S': 0x01, 'T': 0x11,
    'U': 0x20, 'V': 0x09, 'W': 0x0D, 'X': 0x07, 'Y': 0x10,
    'Z': 0x06,
    # 数字
    '0': 0x1D, '1': 0x12, '2': 0x13, '3': 0x14, '4': 0x15,
    '5': 0x17, '6': 0x16, '7': 0x1A, '8': 0x1C, '9': 0x19,
    # 功能键
    'RETURN': 0x24, 'TAB': 0x30, 'SPACE': 0x31, 'DELETE': 0x33,
    'ESCAPE': 0x35, 'ENTER': 0x4C,
    # 修饰键
    'SHIFT': 0x38, 'CONTROL': 0x3B, 'OPTION': 0x3A, 'COMMAND': 0x37,
    'CAPSLOCK': 0x39,
    # 方向键
    'UP': 0x7E, 'DOWN': 0x7D, 'LEFT': 0x7B, 'RIGHT': 0x7C,
}

# 修饰键 Flag
# 这些值来自 CGEvent.h
MODIFIER_FLAGS = {
    'SHIFT': 0x00020000,
    'CONTROL': 0x00040000,
    'OPTION': 0x00080000,
    'COMMAND': 0x00100000,
    'CAPSLOCK': 0x00010000,
}


def _parse_key_combo(combo: str) -> tuple[int, int]:
    """
    解析按键组合字符串，返回 (键码, 修饰键flag)
    
    Examples:
        "A"        → (0x00, 0)
        "Shift+A"  → (0x00, 0x00020000)
        "Cmd+Q"    → (0x0C, 0x00100000)
        "Ctrl+Alt+Del" → (0x33, 0x00040000 | 0x00080000)
    """
    parts = combo.upper().split('+')
    
    flags = 0
    key_name = parts[-1]
    
    for mod in parts[:-1]:
        if mod in MODIFIER_FLAGS:
            flags |= MODIFIER_FLAGS[mod]
    
    key_code = KEYCODE_MAP.get(key_name.upper())
    if key_code is None and len(key_name) == 1:
        # 单个字符的 fallback: 尝试大写字母
        key_code = KEYCODE_MAP.get(key_name.upper())
    
    if key_code is None:
        raise ValueError(f"Unknown key: {key_name}")
    
    return key_code, flags


def press_key(key_combo: str):
    """按下并释放一个按键（或组合键）"""
    key_code, flags = _parse_key_combo(key_combo)
    # 检测单个大写字母，自动添加 Shift（macOS 默认无 Shift 输出小写）
    if len(key_combo) == 1 and key_combo.isupper():
        flags = MODIFIER_FLAGS['SHIFT']
    _press_key_down(key_code, flags)
    _release_key(key_code)
    logger.debug(f"Key pressed: {key_combo} (code=0x{key_code:02X}, flags=0x{flags:08X})")


def press_key_down(key_combo: str):
    """仅按下（不释放）"""
    key_code, flags = _parse_key_combo(key_combo)
    _press_key_down(key_code, flags)


def release_key(key_combo: str):
    """释放按键"""
    key_code, flags = _parse_key_combo(key_combo)
    _release_key(key_code)


def _press_key_down(key_code: int, flags: int = 0):
    if not HAVE_QUARTZ:
        logger.info(f"[SIMULATE] KeyDown: code=0x{key_code:02X}, flags=0x{flags:08X}")
        return
    event = CGEventCreateKeyboardEvent(None, key_code, True)
    CGEventSetFlags(event, flags)
    CGEventPost(kCGHIDEventTap, event)


def _release_key(key_code: int):
    if not HAVE_QUARTZ:
        logger.info(f"[SIMULATE] KeyUp: code=0x{key_code:02X}")
        return
    event = CGEventCreateKeyboardEvent(None, key_code, False)
    CGEventPost(kCGHIDEventTap, event)


def is_accessibility_enabled() -> bool:
    """检查辅助功能权限是否已授权"""
    # 使用 AXIsProcessTrusted() 是可靠的方法
    # CGEventPost 在权限不足时静默失败，不能依赖 try/except
    try:
        from ApplicationServices import AXIsProcessTrusted
        return AXIsProcessTrusted()
    except ImportError:
        return False


def type_text(text: str):
    """输入一段文本（逐个字符按键）"""
    for char in text:
        upper = char.upper()
        needs_shift = char.isupper() or char in '~!@#$%^&*()_+{}|:"<>?'
        
        if needs_shift:
            press_key(f"Shift+{upper}")
        else:
            press_key(upper)
```

// ── WebSocket 连接 ──
let ws = null;
let reconnectTimer = null;
let clientId = null;
let pingInterval = null;

function connect() {
    const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
    const wsUrl = `${protocol}//${window.location.host}/ws`;
    
    setStatus('connecting', '正在连接...');
    
    ws = new WebSocket(wsUrl);
    
    ws.onopen = () => {
        setStatus('connected', '已连接');
        startPing();
        // 请求当前布局
        send({ type: 'profile_request' });
    };
    
    ws.onclose = () => {
        setStatus('disconnected', '已断开');
        stopPing();
        scheduleReconnect();
    };
    
    ws.onerror = () => {
        // onclose 会随后触发
    };
    
    ws.onmessage = (event) => {
        const msg = JSON.parse(event.data);
        handleMessage(msg);
    };
}

function send(data) {
    if (ws && ws.readyState === WebSocket.OPEN) {
        ws.send(JSON.stringify(data));
    }
}

function scheduleReconnect() {
    if (reconnectTimer) return;
    setStatus('connecting', '正在重连...');
    reconnectTimer = setTimeout(() => {
        reconnectTimer = null;
        connect();
    }, 2000);
}

function startPing() {
    pingInterval = setInterval(() => {
        send({ type: 'ping' });
    }, 10000);
}

function stopPing() {
    if (pingInterval) {
        clearInterval(pingInterval);
        pingInterval = null;
    }
}

// ── 消息处理 ──
function handleMessage(msg) {
    switch (msg.type) {
        case 'connected':
            clientId = msg.client_id;
            document.getElementById('server-info').textContent = `ID: ${clientId}`;
            break;
            
        case 'pong':
            break;
            
        case 'profile':
            // 隐藏加载提示
            const hint = document.getElementById('loading-hint');
            if (hint) hint.style.display = 'none';
            renderProfile(msg.profile);
            break;
            
        case 'key_feedback':
            onKeyFeedback(msg.key, msg.action);
            break;
            
        case 'error':
            console.error('Server error:', msg.message);
            break;
    }
}

    el.textContent = text;

let currentProfile = null;

function renderProfile(profile) {
    currentProfile = profile;
    const grid = document.getElementById('keyboard-grid');
    grid.innerHTML = '';
    
    if (!profile || !profile.pages || profile.pages.length === 0) return;
    
    const page = profile.pages[0];
    const keys = page.keys || [];
    
    // 设置背景
    if (profile.settings?.background) {
        document.getElementById('app').style.background = profile.settings.background;
    }
    
    keys.forEach(keyData => {
        const el = document.createElement('div');
        el.className = 'key';
        el.dataset.keyId = keyData.id;
        el.dataset.keyValue = keyData.value;
        el.dataset.keyType = keyData.type || 'keyboard';
        
        // 网格位置 (基于 gridSize=12)
        el.style.gridColumn = `${keyData.x + 1} / span ${keyData.width}`;
        el.style.gridRow = `${keyData.y + 1} / span ${keyData.height}`;
        
        // 自定义颜色
        if (keyData.display?.color) {
            el.style.background = keyData.display.color;
        }
        
        // 标签
        el.textContent = keyData.display?.label || keyData.value;
        
        // 触摸事件
        el.addEventListener('touchstart', (e) => {
            e.preventDefault();
            el.classList.add('pressed');
            send({
                type: 'key_press',
                key: keyData.value,
                action: 'down'
            });
        }, { passive: false });
        
        el.addEventListener('touchend', (e) => {
            e.preventDefault();
            el.classList.remove('pressed');
            send({
                type: 'key_press',
                key: keyData.value,
                action: 'up'
            });
        }, { passive: false });
        
        // 鼠标支持（PC 调试用）
        el.addEventListener('mousedown', () => {
            el.classList.add('pressed');
            send({ type: 'key_press', key: keyData.value, action: 'down' });
        });
        el.addEventListener('mouseup', () => {
            el.classList.remove('pressed');
            send({ type: 'key_press', key: keyData.value, action: 'up' });
        });
        el.addEventListener('mouseleave', () => {
            el.classList.remove('pressed');
            send({ type: 'key_press', key: keyData.value, action: 'up' });
        });
        
        grid.appendChild(el);
    });
}

```python
# server/profile_manager.py
import json
import os
from pathlib import Path
from typing import Optional
import shutil
import logging

logger = logging.getLogger("stp.profile")

PROFILES_DIR = Path(__file__).parent / "profiles"
PROFILES_DIR.mkdir(exist_ok=True)


class ProfileManager:
    """Profile 的本地 JSON 文件存储管理器"""

    def __init__(self, profiles_dir: Path = PROFILES_DIR):
        self.dir = profiles_dir

    def list_profiles(self) -> list[dict]:
        """列出所有 Profile（摘要信息）"""
        profiles = []
        for f in sorted(self.dir.glob("*.json")):
            try:
                data = json.loads(f.read_text())
                profiles.append({
                    "name": data.get("profileName", f.stem),
                    "filename": f.name,
                    "version": data.get("version", "1.0"),
                    "pageCount": len(data.get("pages", [])),
                    "keyCount": sum(len(p.get("keys", [])) for p in data.get("pages", []))
                })
            except Exception as e:
                logger.warning(f"Failed to load profile {f.name}: {e}")
        return profiles

    def get_profile(self, filename: str) -> Optional[dict]:
        path = self.dir / filename
        if not path.exists():
            return None
        return json.loads(path.read_text())

    def save_profile(self, profile: dict, filename: Optional[str] = None) -> str:
        if not filename:
            filename = f"{profile.get('profileName', 'untitled')}.json"
        if not filename.endswith(".json"):
            filename += ".json"
        path = self.dir / filename
        path.write_text(json.dumps(profile, indent=2, ensure_ascii=False))
        logger.info(f"Profile saved: {filename}")
        return filename

    def delete_profile(self, filename: str) -> bool:
        path = self.dir / filename
        if path.exists():
            path.unlink()
            logger.info(f"Profile deleted: {filename}")
            return True
        return False
```

```python
"""
macOS 前台窗口监控
使用 NSWorkspace 通知监听前台应用切换
注意: observer 必须继承 NSObject，不能是普通 Python 对象
"""
import logging
import threading

logger = logging.getLogger("stp.window_watcher")

try:
    from AppKit import NSWorkspace
    from Foundation import NSObject
    from PyObjCTools import AppHelper
    import objc
    HAVE_PYOBJC = True
except ImportError:
    HAVE_PYOBJC = False
    # 桩类，确保导入不报错
    class NSObject: pass


class AppObserver(NSObject):
    """PyObjC 通知观察者 — 必须继承 NSObject"""

    def initWithCallback_(self, callback):
        self = objc.super(AppObserver, self).init()
        if self:
            self.callback = callback
        return self

    def appActivated_(self, notification):
        app = notification.userInfo().get("NSWorkspaceApplicationKey")
        if app and hasattr(self, 'callback') and self.callback:
            info = {
                "name": app.localizedName(),
                "bundle_id": app.bundleIdentifier()
            }
            logger.info(f"App switched: {info['name']} ({info['bundle_id']})")
            self.callback(info)


class WindowWatcher:
    """监听 macOS 前台窗口变化，触发回调"""

    def __init__(self, on_app_changed=None):
        self.on_app_changed = on_app_changed
        self._running = False
        self._thread = None
        self._observer = None

    def start(self):
        if not HAVE_PYOBJC:
            logger.warning("PyObjC not available — window monitoring disabled")
            return

        self._running = True
        self._thread = threading.Thread(target=self._run, daemon=True)
        self._thread.start()
        logger.info("Window watcher started")

    def stop(self):
        self._running = False

    def _run(self):
        workspace = NSWorkspace.sharedWorkspace()
        notification_center = workspace.notificationCenter()

        # 创建 NSObject 子类实例作为 observer
        self._observer = AppObserver.alloc().initWithCallback_(self.on_app_changed)

        notification_center.addObserver_selector_name_object_(
            self._observer,
            "appActivated:",
            "NSWorkspaceDidActivateApplicationNotification",
            None
        )

        # 保持事件循环
        AppHelper.runConsoleEventLoop(installInterrupt=False)


def get_frontmost_app() -> dict:
    """获取当前最前应用"""
    if not HAVE_PYOBJC:
        return {"name": "unknown", "bundle_id": "unknown"}
    app = NSWorkspace.sharedWorkspace().frontmostApplication()
    return {
        "name": app.localizedName(),
        "bundle_id": app.bundleIdentifier()
    }
```
