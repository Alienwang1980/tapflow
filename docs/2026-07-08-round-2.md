### B3 — iPad 触控卡键修复

**当前代码** (`touchend`):
```js
document.addEventListener("touchend",e=>{
  e.preventDefault();
  for(let i=0;i<e.changedTouches.length;i++){
    const t=e.changedTouches[i],
          el=document.elementFromPoint(t.clientX,t.clientY); // ← 手指位置下可能已无 key-btn
    if(el?.classList.contains("key-btn")){
      // 只处理手指仍在 button 上方的情况
    }
  }
},{passive:false});
```

**问题**: 手指滑出 button 后 `elementFromPoint` 不再是 key-btn → 不会发送 `up` → key 卡在按下状态。

**修改为**:
```js
document.addEventListener("touchend",e=>{
  e.preventDefault();
  for(let i=0;i<e.changedTouches.length;i++){
    const t=e.changedTouches[i];
    // 遍历 active 中所有 key，找到对应 touch 的 key 并释放
    const el=document.elementFromPoint(t.clientX,t.clientY);
    const kid=el?.dataset?.keyId;
    // 方案: touchend 时释放 active 中所有匹配的 key
    active.forEach(activeKid=>{
      const b=document.querySelector('[data-key-id="'+activeKid+'"]');
      if(b)b.classList.remove("active");
      const pg=profile?.pages.find(p=>p.id===activePage),
            kd=pg?.keys.find(k=>k.id===activeKid);
      if(kd&&ws&&ws.readyState===WebSocket.OPEN)
        ws.send(JSON.stringify({type:"key",key:kd.value,action:"up"}));
    });
    active.clear();
  }
},{passive:false});
```

同时新增 `touchcancel`:
```js
document.addEventListener("touchcancel",e=>{
  e.preventDefault();
  // 同 touchend 逻辑 — 释放所有 active key
  active.forEach(activeKid=>{ /* 同上 */ });
  active.clear();
},{passive:false});
```

**影响评估**: 多点触控场景下，用户抬起一根手指会释放**所有** active key。对于同时按住多个 modifier 键的场景，需要更精细的处理——但当前 MVP 阶段先简化处理。

```js
let captureMode=false;
function startCapture(){
  captureMode=true;
  document.getElementById("kv").placeholder="按任意键...";
  document.getElementById("kv").focus();
}
document.addEventListener("keydown",e=>{
  if(!captureMode)return;
  e.preventDefault();
  // 跳过纯修饰键
  if(["Shift","Control","Alt","Meta"].includes(e.key))return;
  // 构建 combo 字符串
  let combo="";
  if(e.metaKey)combo+="COMMAND+";
  if(e.ctrlKey)combo+="CONTROL+";
  if(e.altKey)combo+="OPTION+";
  if(e.shiftKey&&e.key.length>1)combo+="SHIFT+";
  combo+=e.key.toUpperCase();
  // 填入输入框
  document.getElementById("kv").value=combo;
  upk("value",combo);
  captureMode=false;
  document.getElementById("kv").placeholder="";
});
```
